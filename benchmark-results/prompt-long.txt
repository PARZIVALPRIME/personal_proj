/* kv16.h -- a KV cache in float16: half the RAM of float32. Each value is clamped to
 * float16's range (+-65504: a value past it is the largest one, never an infinity) and
 * rounded to the nearest float16, ties to even, the IEEE rounding: the CPU's (F16C, or the
 * integer path below), a CUDA kernel's (__float2half_rn) and a Vulkan shader's (the same
 * integer arithmetic as kv16_round_bits) give the same bits, and a row read back widens
 * to the float it was rounded to.
 *
 * Float16, not bfloat16: three more bits of mantissa. On Muse Spark's tiny oracle, whose
 * heads' outputs are normalised (a head whose values cancel magnifies their rounding), K
 * and V in bf16 moved the teacher-forced NLL from 2.0790 to 2.2305 nats a token, in f16
 * to 2.0752.
 *
 * kv16_dot_lanes is dot_f32_lanes (matmul_f32.h) over a float16 row: the same lanes in
 * the same order, so a score is the same bits whether its key comes from the cache or
 * from a step's rounded scratch. */
#ifndef COLI_KV16_H
#define COLI_KV16_H
#include <stdint.h>
#include <string.h>
#include "matmul_f32.h"   /* MATMUL_F32_LANES, MATMUL_F32_MADD */

#define KV16_MAX 65504.0f

/* the float32 bits of x rounded to float16's precision (x within +-65504): the integer
 * reference the Vulkan shader (chain_ew.comp, op 11) follows line for line */
static inline uint32_t kv16_round_bits(uint32_t u) {
    uint32_t s = u & 0x80000000u, a = u & 0x7fffffffu;
    if (a > 0x7f800000u) return u | 0x400000u;            /* a NaN stays one */
    if (a >= 0x38800000u)                                  /* float16's normals: 10 bits of mantissa */
        return s | ((a + 0xfffu + ((a >> 13) & 1u)) & 0xffffe000u);
    /* below 2^-14, float16's subnormals: multiples of 2^-24 */
    uint32_t e = a >> 23, m = (a & 0x7fffffu) | 0x800000u;
    if (e < 102) return s;                                 /* under 2^-25: zero (2^-25 itself a tie, to even) */
    uint32_t sh = 126 - e;                                 /* 14 .. 24: the value is m * 2^-(sh + 24) */
    uint32_t n = (m + (1u << (sh - 1)) - 1u + ((m >> sh) & 1u)) >> sh;
    float f = (float)n * 5.9604644775390625e-8f;           /* n * 2^-24, exact */
    uint32_t r; memcpy(&r, &f, 4);
    return s | r;
}
static inline float kv16_clamp(float x) { return x > KV16_MAX ? KV16_MAX : x < -KV16_MAX ? -KV16_MAX : x; }

#if defined(__FLT16_MAX__) && !defined(KV16_INTEGER)   /* KV16_INTEGER: the integer path (tests) */
typedef _Float16 kv16_t;
static inline uint16_t kv16_from(float f) { kv16_t h = (kv16_t)kv16_clamp(f); uint16_t b; memcpy(&b, &h, 2); return b; }
static inline float kv16_to(uint16_t b) { kv16_t h; memcpy(&h, &b, 2); return (float)h; }
#else
/* no _Float16 in this compiler: the bits by integer arithmetic */
static inline uint16_t kv16_from(float f) {
    f = kv16_clamp(f);
    uint32_t u; memcpy(&u, &f, 4);
    u = kv16_round_bits(u);
    uint32_t s = (u >> 16) & 0x8000u, a = u & 0x7fffffffu;
    if (a > 0x7f800000u) return (uint16_t)(s | 0x7e00u);
    if (a < 0x38800000u) { float g; memcpy(&g, &a, 4); return (uint16_t)(s | (uint32_t)(g * 16777216.0f)); }   /* n * 2^-24 */
    return (uint16_t)(s | (((a >> 23) - 112u) << 10) | ((a >> 13) & 0x3ffu));
}
static inline float kv16_to(uint16_t b) {
    uint32_t s = ((uint32_t)b & 0x8000u) << 16, e = (b >> 10) & 0x1fu, m = b & 0x3ffu, u;
    if (e == 0) { float g = (float)m * 5.9604644775390625e-8f; memcpy(&u, &g, 4); u |= s; }
    else if (e == 31) u = s | 0x7f800000u | (m << 13);
    else u = s | ((e + 112u) << 23) | (m << 13);
    float f; memcpy(&f, &u, 4);
    return f;
}
#endif

/* gcc does not vectorize the _Float16 casts: with F16C the three loops below convert eight
 * values at a time by the instructions those casts compile to (round to nearest even), the
 * clamp's operands ordered so a NaN passes through as kv16_clamp lets it: the same bits.
 * Before, kv16_round took about 0.2 ms of an OLMoE decode token (16 layers' K and V rows
 * down from the Vulkan chain) and about 20 ms of a 3K-token prompt. */
#if defined(__F16C__) && defined(__AVX__) && defined(__FLT16_MAX__) && !defined(KV16_INTEGER)
#include <immintrin.h>
#define KV16_CVT8 1
static inline __m256 kv16_clamp8(__m256 v) {
    return _mm256_max_ps(_mm256_set1_ps(-KV16_MAX), _mm256_min_ps(_mm256_set1_ps(KV16_MAX), v));
}
#endif
/* x rounded in place to the floats its float16 holds */
static inline void kv16_round(float *x, int64_t n) {
    int64_t i = 0;
#ifdef KV16_CVT8
    for (; i + 8 <= n; i += 8)
        _mm256_storeu_ps(x + i, _mm256_cvtph_ps(_mm256_cvtps_ph(kv16_clamp8(_mm256_loadu_ps(x + i)), _MM_FROUND_TO_NEAREST_INT)));
#endif
    for (; i < n; i++) x[i] = kv16_to(kv16_from(x[i]));
}
static inline void kv16_pack(uint16_t *d, const float *s, int64_t n) {
    int64_t i = 0;
#ifdef KV16_CVT8
    for (; i + 8 <= n; i += 8)
        _mm_storeu_si128((__m128i *)(d + i), _mm256_cvtps_ph(kv16_clamp8(_mm256_loadu_ps(s + i)), _MM_FROUND_TO_NEAREST_INT));
#endif
    for (; i < n; i++) d[i] = kv16_from(s[i]);
}
static inline void kv16_unpack(float *d, const uint16_t *s, int64_t n) {
    int64_t i = 0;
#ifdef KV16_CVT8
    for (; i + 8 <= n; i += 8) _mm256_storeu_ps(d + i, _mm256_cvtph_ps(_mm_loadu_si128((const __m128i *)(s + i))));
#endif
    for (; i < n; i++) d[i] = kv16_to(s[i]);
}

/* gcc does not vectorize a float16 -> float conversion without AVX512-FP16: the two loops
 * attention runs over every position, with F16C's conversion by hand, lane for lane the
 * scalar ones (lane j of 32 a fused multiply-add chain; the same halving tree after) */
#if defined(__F16C__) && defined(__AVX__) && defined(__FMA__) && MATMUL_F32_LANES == 32
#include <immintrin.h>
#define KV16_F16C 1
#endif

static inline float kv16_dot_lanes(const float *x, const uint16_t *w, int I) {
    float a[MATMUL_F32_LANES] = {0};
    int i = 0;
#ifdef KV16_F16C
    __m256 c0 = _mm256_setzero_ps(), c1 = c0, c2 = c0, c3 = c0;
    for (; i + 32 <= I; i += 32) {
        c0 = _mm256_fmadd_ps(_mm256_loadu_ps(x + i),      _mm256_cvtph_ps(_mm_loadu_si128((const __m128i *)(w + i))),      c0);
        c1 = _mm256_fmadd_ps(_mm256_loadu_ps(x + i + 8),  _mm256_cvtph_ps(_mm_loadu_si128((const __m128i *)(w + i + 8))),  c1);
        c2 = _mm256_fmadd_ps(_mm256_loadu_ps(x + i + 16), _mm256_cvtph_ps(_mm_loadu_si128((const __m128i *)(w + i + 16))), c2);
        c3 = _mm256_fmadd_ps(_mm256_loadu_ps(x + i + 24), _mm256_cvtph_ps(_mm_loadu_si128((const __m128i *)(w + i + 24))), c3);
    }
    _mm256_storeu_ps(a, c0); _mm256_storeu_ps(a + 8, c1); _mm256_storeu_ps(a + 16, c2); _mm256_storeu_ps(a + 24, c3);
#else
    for (; i + MATMUL_F32_LANES <= I; i += MATMUL_F32_LANES)
        for (int j = 0; j < MATMUL_F32_LANES; j++) a[j] = MATMUL_F32_MADD(a[j], x[i+j], kv16_to(w[i+j]));
#endif
    for (int j = 0; i + j < I; j++) a[j] = MATMUL_F32_MADD(a[j], x[i+j], kv16_to(w[i+j]));
    for (int h = MATMUL_F32_LANES/2; h > 0; h >>= 1)
        for (int j = 0; j < h; j++) a[j] += a[j+h];
    return a[0];
}
/* y += a * x, the float16 row widened (the float row's own loop: y + a*x, fused). The
 * fusion is spelled out (MATMUL_F32_MADD) rather than left to the compiler, which fuses
 * y += a*x only from -O2 on: at -O1 (the ASan build) the tail would round twice while the
 * F16C body does not. */
static inline void kv16_axpy(float *y, float a, const uint16_t *x, int n) {
    int i = 0;
#ifdef KV16_F16C
    __m256 va = _mm256_set1_ps(a);
    for (; i + 8 <= n; i += 8)
        _mm256_storeu_ps(y + i, _mm256_fmadd_ps(va, _mm256_cvtph_ps(_mm_loadu_si128((const __m128i *)(x + i))), _mm256_loadu_ps(y + i)));
#endif
    for (; i < n; i++) y[i] = MATMUL_F32_MADD(y[i], a, kv16_to(x[i]));
}
#endif
/* pin_pool.h — piu fotografie dello stato, non una sola.
 *
 * IL PROBLEMA. Una fotografia sola (SUBMIT pin=1) copre UN prefisso. Chi
 * punteggia un insieme chiuso ne ha due annidati: le istruzioni, condivise da
 * mille richieste, e istruzioni+domanda, condivise dalle alternative di quella
 * richiesta. Con una sola si e' costretti a scegliere: o si tengono le
 * istruzioni e ogni alternativa fa rileggere la domanda da capo, o si tiene la
 * domanda e le istruzioni si rileggono a ogni item.
 *
 * Misurato su qwen36 (22 GB), 4 item, istruzioni da 121 token, 3 opzioni:
 * con una fotografia sola jev macina 124 token per item invece di 46, cioe'
 * tre volte la domanda, e passa da 126,6 s a item contro i 60,1 della
 * generazione. Non e' un limite del metodo, e' questo pezzo che manca.
 *
 * LA FORMA. Un pugno di scatti; si rimette SEMPRE il piu' profondo che sia un
 * prefisso stretto del prompt nuovo. Nessuna chiave nuova sul filo: il client
 * manda pin=1 dove gli serve un punto di ritorno, e il motore li tiene tutti
 * finche' c'e' posto. kimi_k3.c fa gia' cosi' con i suoi checkpoint (di li'
 * viene l'idea di tenere il piu' profondo valido); qui la parte che non
 * dipende dal motore sta in un posto solo.
 *
 * COSA C'E' DENTRO E COSA NO. Qui stanno gli id, i logit finali e la
 * contabilita' degli slot. Lo stato ricorrente NO: e' diverso in ogni motore
 * (DeltaNet, SSM, KDA, convoluzioni corte) e alcuni non ne hanno affatto. Ogni
 * motore se lo attacca a `state` e se lo copia da se'.
 *
 * QUELLO CHE QUESTO FILE NON PUO' SAPERE. Che le righe K/V di quelle posizioni
 * siano ancora vive. Non entrano nella fotografia, restano dove sono, e i
 * banchi possono essere stati ributtati nel frattempo. Chi rimette uno scatto
 * DEVE chiederlo a kv_prefix_holds() prima di fidarsi. */
#ifndef COLI_PIN_POOL_H
#define COLI_PIN_POOL_H

#include <stdlib.h>
#include <string.h>

#ifndef COLI_PIN_SLOTS_MAX
#define COLI_PIN_SLOTS_MAX 8
#endif

typedef struct {
    int   *ids;      /* copia propria degli id che lo scatto copre */
    int    len;      /* quante posizioni */
    int    cap;      /* capienza di ids */
    float *logit;    /* logit dell'ultima posizione coperta: predice il primo token fresco */
    void  *state;    /* blob del motore (stato ricorrente), opaco qui */
    unsigned long long used;   /* marca LRU */
} ColiPin;

typedef struct {
    ColiPin slot[COLI_PIN_SLOTS_MAX];
    int n;                       /* slot in uso, 0 = spento */
    int vocab;
    unsigned long long clock;
} ColiPinPool;

/* Quanti scatti tenere. Default 4: bastano istruzioni + domanda con margine.
 * Su un motore con stato ricorrente grosso ogni scatto costa (qwen36: decine
 * di MB), quindi si puo' abbassare senza toccare il codice. */
static inline int coli_pin_slots_wanted(void) {
    const char *e = getenv("COLI_PIN_SLOTS");
    int n = e ? atoi(e) : 4;
    if (n < 0) n = 0;
    if (n > COLI_PIN_SLOTS_MAX) n = COLI_PIN_SLOTS_MAX;
    return n;
}

static inline void coli_pin_pool_init(ColiPinPool *p, int vocab) {
    if (!p || p->n) return;
    memset(p, 0, sizeof(*p));
    p->n = coli_pin_slots_wanted();
    p->vocab = vocab;
}

/* Lo scatto piu' profondo che sia un prefisso STRETTO di questo prompt.
 *
 * Stretto perche' serve almeno un token nuovo: un prompt uguale allo scatto
 * vorrebbe dire riavvolgere fino a se stesso e non lascerebbe niente da
 * macinare, quindi nessun logit da cui ripartire. Torna l'indice o -1. */
static inline int coli_pin_best(const ColiPinPool *p, const int *ids, int n) {
    if (!p || !ids || n < 1) return -1;
    int best = -1;
    for (int s = 0; s < p->n; s++) {
        const ColiPin *k = &p->slot[s];
        if (k->len < 1 || !k->ids || k->len >= n) continue;
        if (memcmp(k->ids, ids, (size_t)k->len * sizeof(int)) != 0) continue;
        if (best < 0 || k->len > p->slot[best].len) best = s;
    }
    return best;
}

/* Dove mettere lo scatto nuovo: uno slot libero, oppure lo stesso prefisso se
 * c'e' gia' (rinfrescarlo, non duplicarlo), oppure il meno usato di recente. */
static inline int coli_pin_victim(ColiPinPool *p, const int *ids, int n) {
    if (!p || p->n < 1) return -1;
    for (int s = 0; s < p->n; s++)
        if (p->slot[s].len == n && p->slot[s].ids &&
            memcmp(p->slot[s].ids, ids, (size_t)n * sizeof(int)) == 0) return s;
    for (int s = 0; s < p->n; s++) if (p->slot[s].len < 1) return s;
    int lru = 0;
    for (int s = 1; s < p->n; s++) if (p->slot[s].used < p->slot[lru].used) lru = s;
    return lru;
}

/* Scrive id e logit nello slot. Torna lo slot, o NULL se non c'e' memoria: la
 * fotografia e' un'ottimizzazione e non deve MAI essere il motivo di un
 * errore, quindi chi chiama tira dritto. `state` non viene toccato: se lo
 * slot ne portava uno, il chiamante lo riusa o lo libera lui. */
static inline ColiPin *coli_pin_store(ColiPinPool *p, const int *ids, int n,
                                      const float *logit) {
    if (!p || p->n < 1 || !ids || n < 1) return NULL;
    int s = coli_pin_victim(p, ids, n);
    if (s < 0) return NULL;
    ColiPin *k = &p->slot[s];
    if (k->cap < n) {
        int *grown = (int *)realloc(k->ids, (size_t)n * sizeof(int));
        if (!grown) { k->len = 0; return NULL; }
        k->ids = grown; k->cap = n;
    }
    if (logit && p->vocab > 0) {
        if (!k->logit) k->logit = (float *)malloc((size_t)p->vocab * sizeof(float));
        if (!k->logit) { k->len = 0; return NULL; }
        memcpy(k->logit, logit, (size_t)p->vocab * sizeof(float));
    }
    memcpy(k->ids, ids, (size_t)n * sizeof(int));
    k->len = n;
    k->used = ++p->clock;
    return k;
}

static inline void coli_pin_touch(ColiPinPool *p, int s) {
    if (p && s >= 0 && s < p->n) p->slot[s].used = ++p->clock;
}

/* Butta ogni scatto. Si chiama quando lo stato che descrivono non esiste piu'
 * (banchi K/V riallocati, sessione chiusa): uno scatto orfano rimesso a posto
 * risponderebbe da posizioni che non ci sono, e lo farebbe in silenzio. */
static inline void coli_pin_pool_clear(ColiPinPool *p, void (*free_state)(void *)) {
    if (!p) return;
    for (int s = 0; s < p->n; s++) {
        if (free_state && p->slot[s].state) free_state(p->slot[s].state);
        p->slot[s].state = NULL;
        p->slot[s].len = 0;
    }
}

#endif /* COLI_PIN_POOL_H */
